#include <Wire.h>

const int MPU_ADDR = 0x68; // Dirección I2C predeterminada del MPU-6050

int16_t rawAcX, rawAcY, rawAcZ;
float AcX, AcY, AcZ;
float pitch, roll;

void setup() {
  Serial.begin(115200);
  Wire.begin();
  
  // Despertar el MPU-6050
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B); // Registro PWR_MGMT_1
  Wire.write(0);    // Poner en 0 para encender
  Wire.endTransmission(true);

  Serial.println("Calculando Pitch y Roll con MPU-6050...");
}

void loop() {
  // 1. Solicitar datos del acelerómetro comenzando en 0x3B (ACCEL_XOUT_H)
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);
  Wire.endTransmission(false);
  Wire.requestFrom(MPU_ADDR, 6, true);
  
  // 2. Leer los 6 bytes de datos del acelerómetro
  rawAcX = Wire.read() << 8 | Wire.read(); 
  rawAcY = Wire.read() << 8 | Wire.read(); 
  rawAcZ = Wire.read() << 8 | Wire.read(); 

  // 3. Convertir lecturas crudas a unidades de gravedad (g)
  // Con escala por defecto de +/-2g, dividimos entre 16384.0
  AcX = rawAcX / 16384.0;
  AcY = rawAcY / 16384.0;
  AcZ = rawAcZ / 16384.0;

  // 4. Calcular los ángulos Pitch y Roll en grados
  roll  = atan2(AcY, sqrt(AcX * AcX + AcZ * AcZ)) * 180.0 / M_PI;
  pitch = atan2(-AcX, sqrt(AcY * AcY + AcZ * AcZ)) * 180.0 / M_PI;

  // 5. Imprimir los resultados
  //Serial.print("Roll (Eje X): ");
  Serial.print(roll);
  Serial.print(",");
  //Serial.print("° \t| Pitch (Eje Y): ");
  Serial.print(pitch);
  //Serial.println("°");

  delay(20);
}
